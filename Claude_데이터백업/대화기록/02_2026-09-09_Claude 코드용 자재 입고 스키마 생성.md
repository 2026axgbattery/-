## 1. SQL (PostgreSQL 기준)

```sql
CREATE TABLE inbound_receipts (
  id             BIGSERIAL PRIMARY KEY,
  seq_no         INTEGER,                        -- No. (원본 순번)
  received_at    DATE        NOT NULL,           -- 입고일자
  material_code  TEXT        NOT NULL,           -- 자재코드
  material_name  TEXT        NOT NULL,           -- 자재명
  spec           TEXT,                           -- 규격
  category_major TEXT,                           -- 대분류
  category_mid   TEXT,                           -- 중분류
  quantity       NUMERIC(18,3) NOT NULL CHECK (quantity >= 0),  -- 입고수량
  status         TEXT        NOT NULL,           -- 상태
  po_no          TEXT,                           -- PO No.
  created_at     TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_inbound_received_at   ON inbound_receipts (received_at);
CREATE INDEX idx_inbound_material_code ON inbound_receipts (material_code);
CREATE INDEX idx_inbound_po_no         ON inbound_receipts (po_no);

-- 중복 입고 방지가 필요하면
CREATE UNIQUE INDEX uq_inbound_row
  ON inbound_receipts (po_no, material_code, received_at, seq_no);
```

SQLite면 `BIGSERIAL` → `INTEGER PRIMARY KEY AUTOINCREMENT`, `NUMERIC(18,3)` → `REAL`, `TIMESTAMPTZ` → `TEXT`로 바꾸면 그대로 씁니다.

## 2. Python (Pydantic v2)

```python
from datetime import date
from decimal import Decimal
from enum import Enum
from pydantic import BaseModel, Field

class ReceiptStatus(str, Enum):
    NORMAL = "정상"
    HOLD = "보류"
    RETURNED = "반품"
    CANCELED = "취소"

class InboundReceipt(BaseModel):
    seq_no: int | None = Field(None, alias="No.")
    received_at: date = Field(alias="입고일자")
    material_code: str = Field(alias="자재코드", min_length=1)
    material_name: str = Field(alias="자재명", min_length=1)
    spec: str | None = Field(None, alias="규격")
    category_major: str | None = Field(None, alias="대분류")
    category_mid: str | None = Field(None, alias="중분류")
    quantity: Decimal = Field(alias="입고수량", ge=0)
    status: ReceiptStatus = Field(alias="상태")
    po_no: str | None = Field(None, alias="PO No.")

    model_config = {"populate_by_name": True}
```

`alias`를 한글 헤더로 잡아둬서 엑셀/CSV를 `InboundReceipt(**row)`로 바로 파싱할 수 있습니다.

## 3. TypeScript (Zod)

```ts
import { z } from "zod";

export const ReceiptStatus = z.enum(["정상", "보류", "반품", "취소"]);

export const InboundReceipt = z.object({
  seqNo: z.number().int().nullable(),
  receivedAt: z.coerce.date(),
  materialCode: z.string().min(1),
  materialName: z.string().min(1),
  spec: z.string().nullable(),
  categoryMajor: z.string().nullable(),
  categoryMid: z.string().nullable(),
  quantity: z.coerce.number().nonnegative(),
  status: ReceiptStatus,
  poNo: z.string().nullable(),
});

export type InboundReceipt = z.infer<typeof InboundReceipt>;
```

두 가지만 확인해 주세요. `상태` 값은 제가 임의로 4개를 넣어둔 거라 실제 쓰시는 값으로 바꿔야 하고, 자재코드/자재명/규격/대분류/중분류는 자재별로 고정된 값이라면 `materials` 마스터 테이블로 분리하고 입고 테이블은 `material_code`만 참조하는 게 중복이 없습니다. 원하시면 정규화한 2테이블 버전이나 파일 형태로도 만들어 드릴게요.

## 1. SQL (PostgreSQL 기준)

```sql
-- 자재 마스터 (입고·출고 공통)
CREATE TABLE materials (
  material_code  TEXT PRIMARY KEY,               -- 자재코드
  material_name  TEXT NOT NULL,                  -- 자재명
  spec           TEXT,                           -- 규격
  category_major TEXT,                           -- 대분류
  category_mid   TEXT,                           -- 중분류
  updated_at     TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- 입고 이력
CREATE TABLE inbound_receipts (
  id            BIGSERIAL PRIMARY KEY,
  seq_no        INTEGER,                         -- No.
  received_at   DATE NOT NULL,                   -- 입고일자
  material_code TEXT NOT NULL REFERENCES materials(material_code),
  quantity      NUMERIC(18,3) NOT NULL CHECK (quantity > 0),  -- 입고수량
  status        TEXT NOT NULL,                   -- 상태
  po_no         TEXT,                            -- PO No.
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- 출고 이력
CREATE TABLE outbound_issues (
  id             BIGSERIAL PRIMARY KEY,
  seq_no         INTEGER,                        -- No.
  issued_at      DATE NOT NULL,                  -- 출고일자
  material_code  TEXT NOT NULL REFERENCES materials(material_code),
  quantity       NUMERIC(18,3) NOT NULL CHECK (quantity > 0),  -- 수량
  status         TEXT NOT NULL,                  -- 상태
  issue_reason   TEXT NOT NULL,                  -- 출고사유
  process        TEXT,                           -- 공정
  equipment_name TEXT,                           -- 설비명
  created_at     TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX idx_in_received_at    ON inbound_receipts (received_at);
CREATE INDEX idx_in_material       ON inbound_receipts (material_code);
CREATE INDEX idx_in_po_no          ON inbound_receipts (po_no);
CREATE INDEX idx_out_issued_at     ON outbound_issues (issued_at);
CREATE INDEX idx_out_material      ON outbound_issues (material_code);
CREATE INDEX idx_out_equipment     ON outbound_issues (equipment_name);
CREATE INDEX idx_out_process       ON outbound_issues (process);
```

수량은 양수로만 저장하고 방향은 뷰에서 부호로 처리하면 재고 계산이 간단해집니다.

```sql
CREATE VIEW material_movements AS
SELECT 'IN' AS direction, received_at AS moved_at, material_code,
       quantity AS signed_qty, status, NULL::TEXT AS process, NULL::TEXT AS equipment_name
FROM inbound_receipts
UNION ALL
SELECT 'OUT', issued_at, material_code,
       -quantity, status, process, equipment_name
FROM outbound_issues;

-- 현재고
SELECT material_code, SUM(signed_qty) AS on_hand
FROM material_movements
GROUP BY material_code;
```

## 2. Python (Pydantic v2)

```python
from datetime import date
from decimal import Decimal
from enum import Enum
from pydantic import BaseModel, Field

class InboundStatus(str, Enum):
    NORMAL = "정상"
    HOLD = "보류"
    RETURNED = "반품"
    CANCELED = "취소"

class OutboundStatus(str, Enum):
    NORMAL = "정상"
    CANCELED = "취소"
    RETURNED = "반납"

class IssueReason(str, Enum):
    PRODUCTION = "생산투입"
    MAINTENANCE = "설비보전"
    DEFECT_REPLACE = "불량교체"
    SCRAP = "폐기"
    TRANSFER = "반출"

class Material(BaseModel):
    material_code: str = Field(alias="자재코드", min_length=1)
    material_name: str = Field(alias="자재명", min_length=1)
    spec: str | None = Field(None, alias="규격")
    category_major: str | None = Field(None, alias="대분류")
    category_mid: str | None = Field(None, alias="중분류")

    model_config = {"populate_by_name": True}

class InboundReceipt(Material):
    seq_no: int | None = Field(None, alias="No.")
    received_at: date = Field(alias="입고일자")
    quantity: Decimal = Field(alias="입고수량", gt=0)
    status: InboundStatus = Field(alias="상태")
    po_no: str | None = Field(None, alias="PO No.")

class OutboundIssue(Material):
    seq_no: int | None = Field(None, alias="No.")
    issued_at: date = Field(alias="출고일자")
    quantity: Decimal = Field(alias="수량", gt=0)
    status: OutboundStatus = Field(alias="상태")
    issue_reason: IssueReason = Field(alias="출고사유")
    process: str | None = Field(None, alias="공정")
    equipment_name: str | None = Field(None, alias="설비명")
```

`Material`을 상속시켜 두면 엑셀 원본 행을 그대로 파싱하면서도, DB 적재 시에는 마스터와 이력으로 나눠 넣을 수 있습니다.

## 3. TypeScript (Zod)

```ts
import { z } from "zod";

export const MaterialBase = z.object({
  materialCode: z.string().min(1),
  materialName: z.string().min(1),
  spec: z.string().nullable(),
  categoryMajor: z.string().nullable(),
  categoryMid: z.string().nullable(),
});

export const InboundReceipt = MaterialBase.extend({
  seqNo: z.number().int().nullable(),
  receivedAt: z.coerce.date(),
  quantity: z.coerce.number().positive(),
  status: z.enum(["정상", "보류", "반품", "취소"]),
  poNo: z.string().nullable(),
});

export const OutboundIssue = MaterialBase.extend({
  seqNo: z.number().int().nullable(),
  issuedAt: z.coerce.date(),
  quantity: z.coerce.number().positive(),
  status: z.enum(["정상", "취소", "반납"]),
  issueReason: z.enum(["생산투입", "설비보전", "불량교체", "폐기", "반출"]),
  process: z.string().nullable(),
  equipmentName: z.string().nullable(),
});

export type InboundReceipt = z.infer<typeof InboundReceipt>;
export type OutboundIssue = z.infer<typeof OutboundIssue>;
```

`상태`, `출고사유` enum 값은 제가 임의로 넣은 것이니 실제 쓰시는 값으로 교체해 주세요. 그리고 `공정`과 `설비명`도 정해진 목록이 있다면 `equipments` 마스터로 빼서 `equipment_id`로 참조하는 편이 오타나 표기 흔들림을 막아줍니다.
